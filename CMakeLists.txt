function(mod_options target standard)
    set_target_properties(${target} PROPERTIES CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES
        PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -Werror)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
endfunction()
